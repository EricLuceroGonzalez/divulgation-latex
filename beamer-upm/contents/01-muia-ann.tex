\def\layersep{3cm}
\def\nodeinlayersep{1.5cm}


\begin{frame}[fragile]

  \begin{tikzpicture}[
      shorten >=1pt,-Triangle, draw=black!40,
      node distance = 6mm and 24mm,
      start chain = going below,
      every pin edge/.style = {Triangle-,shorten <=1pt},
      neuron/.style = {circle, fill=#1,
          minimum size=17pt, inner sep=0pt,
          on chain},
      annot/.style = {text width=4em, align=center}
    ]
    % Draw the input layer nodes
    \foreach \i in {1,...,4}
    \node[neuron=green!50,
      pin=180:Input \i] (I-\i)    {$x_{\i}$};
    % Draw the hidden layer nodes
    \node[neuron=blue!50,
      above right=6mm and 24mm of I-1.center] (H-1)     {$x_{1}$};
    \foreach \i [count=\j from 1] in {2,...,5}
    \node[neuron=blue!50,
      below=of H-\j]      (H-\i)    {$x_{\i}$};
    % Draw the output layer node
    \node[neuron=red!50,
      pin= {[pin edge=->]0:Output 1},
      right=of I-1 -| H-1]  (O-1)   {$x_{1}$};
    \foreach \i [count=\j from 1] in {2,...,4}
    \node[neuron=red!50,
      pin= {[pin edge=->]0:Output \j},
      below=of O-\j]        (O-\i)  {$x_{\i}$};
    % Connect input nodes with hidden nodes and 
    %  hiden nodes with output nodes with the output layer
    \foreach \i in {1,...,4}
    \foreach \j in {1,...,5}
      {
        \path (I-\i) edge (H-\j)
        (H-\j) edge (O-\i);
      }
  \end{tikzpicture}
\end{frame}

\begin{frame}[fragile]
  \begin{tikzpicture}[
      shorten >=1pt, -Triangle,
      draw=black!40,
      node distance=6mm and 24mm,
      start chain=going below,
      every pin edge/.style={Triangle-, shorten <=1pt},
      neuron/.style={circle, fill=#1, minimum size=20pt, inner sep=2pt, on chain},
      annot/.style={text width=4em, align=center}
    ]

    % Input layer
    \foreach \i in {1,...,3}
    \node[neuron=green!50, pin=180:Input \i] (I-\i) {$x_{\i}$};
    \node[below=of I-3] (I-dots) {$\vdots$};
    \node[neuron=green!50, below=of I-dots, pin=180:Input 21] (I-21) {$x_{21}$};

    % Hidden layer 1
    \node[neuron=blue!50, above right=4mm and 20mm of I-1] (H1-1) {};
    \node[neuron=blue!50, below=of H1-1] (H1-2) {};
    \node[neuron=blue!50, below=of H1-2] (H1-3) {};
    \node[neuron=blue!50, below=of H1-3] (H1-4) {};
    \node[below=of H1-4] (H1-dots) {$\vdots$};
    \node[neuron=blue!50, below=of H1-dots] (H1-n) {};

    % Hidden layer 2
    \node[neuron=blue!50, right=20mm of H1-1] (H2-1) {};
    \node[neuron=blue!50, below=of H2-1] (H2-2) {};
    \node[neuron=blue!50, below=of H2-2] (H2-3) {};
    \node[neuron=blue!50, below=of H2-3] (H2-4) {};
    \node[below=of H2-4] (H2-dots) {$\vdots$};
    \node[neuron=blue!50, below=of H2-dots] (H2-n) {};

    % Output layer
    \node[neuron=red!50, below right=5mm and 16mm of H2-3, pin={[pin edge=->]0:RainTomorrow (0/1)}] (O-1) {};

    % Connections: Input to Hidden 1
    \foreach \i in {1, 2, 3, 21}
    \foreach \j in {1, 2,3,4, n}
    \path (I-\i) edge (H1-\j);

    % Connections: Hidden 1 to Hidden 2
    \foreach \i in {1, 2,3,4, n}
    \foreach \j in {1, 2,3,4, n}
    \path (H1-\i) edge (H2-\j);

    % Connections: Hidden 2 to Output
    \foreach \i in {1,2,3,4,n}
    \path (H2-\i) edge (O-1);

    % Annotations
    \node[annot, above=of I-1] {Input Layer};
    \node[annot, above=of H1-1] {Hidden 1};
    \node[annot, above=of H2-1] {Hidden 2};
    \node[annot, above=of O-1] {Output};

  \end{tikzpicture}
\end{frame}
