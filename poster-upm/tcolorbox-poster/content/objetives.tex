% objetives.tex
\begin{enumerate}[leftmargin=0.4cm,label=\protect\circled{\arabic*}]
    \item Medir la eficiencia en tiempo y la cantidad de comandos requeridos para realizar tareas comunes en la creación de pósteres utilizando \boxi[colorD]{tcolorbox}.
    \item Comparar la calidad visual y la estructura de los pósteres generados con \boxi[colorD]{tcolorbox} y \boxi{beamerposter}.
    \item Evaluar el grado de complejidad para implementar diseños avanzados y personalizados con \boxi[colorD]{tcolorbox}.
    \item Analizar la curva de aprendizaje y la facilidad de uso de \boxi[colorD]{tcolorbox} para usuarios de diferentes niveles (principiante, intermedio, avanzado).
    \item Determinar el nivel de flexibilidad y libertad creativa que ofrece \boxi[colorD]{tcolorbox} en comparación con \boxi{beamerposter}.
\end{enumerate}
